# E0430: A default method that two visible interfaces supply

An interface can give a method a default body, which every type satisfying
it gets for free. When a type satisfies two interfaces that both default
the same method, a call to it has two candidate bodies and no way to choose
between them, so the call is rejected.

```lugha
interface Named {
    fun name(self): string;
    fun describe(self): string = "named {self.name()}";
}

interface Shown {
    fun name(self): string;
    fun describe(self): string = "shown {self.name()}";
}

struct Circle { r: f64 }

impl Circle {
    fun name(self): string = "circle";
}

fun main() {
    let c = Circle { r: 1.0 };
    println(c.describe()); // error: E0430
}
```

Define the method on the type itself. A type's own method always wins over
a default:

```lugha
interface Named {
    fun name(self): string;
    fun describe(self): string = "named {self.name()}";
}

interface Shown {
    fun name(self): string;
    fun describe(self): string = "shown {self.name()}";
}

struct Circle { r: f64 }

impl Circle {
    fun name(self): string = "circle";
    fun describe(self): string = "a circle of radius {self.r}";
}

fun main() {
    let c = Circle { r: 1.0 };
    println(c.describe());
}
```

Or call it through one interface, where there is only one default:

```lugha
interface Named {
    fun name(self): string;
    fun describe(self): string = "named {self.name()}";
}

interface Shown {
    fun name(self): string;
    fun describe(self): string = "shown {self.name()}";
}

struct Circle { r: f64 }

impl Circle {
    fun name(self): string = "circle";
}

fun main() {
    let n: Named = Circle { r: 1.0 };
    println(n.describe());
}
```
